# Dependencies for mcp_server.py only.
#
# Deliberately separate from requirements.txt: the Cloud Run image that serves
# the REST API does not need fastmcp, and adding it there would grow the image
# for no reason. Install this only where the MCP server actually runs — a
# developer machine for stdio, or the API image as well IF you later mount
# mcp_server.asgi_app() into app.py.
#
#   python3 -m venv .venv && . .venv/bin/activate
#   pip install -r requirements-mcp.txt
#
# Verified against fastmcp 3.4.7. The ASGI accessor has been renamed across
# major versions, which asgi_app() handles with a fallback chain — if you pin a
# different major, re-run the smoke test in the module docstring.
fastmcp>=3.0,<4.0
httpx>=0.27
